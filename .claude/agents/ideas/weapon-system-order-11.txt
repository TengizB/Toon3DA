C. WeaponRoller.java — class

package ge.tbegvadze.toon3d.entity;

import java.util.Random;

/**
 * Rolls weapon level, tier, and abilities at spawn time.
 * Stateless after construction; call rollWeapon() for each new pickup.
 */
public class WeaponRoller {

    private final Random rng;  // seeded from World.runSeed for reproducibility

    public WeaponRoller(long runSeed) {
        this.rng = new Random(runSeed);
    }

    /**
     * Configures a weapon with a rolled level, tier, and abilities appropriate
     * for the given floor depth.
     *
     * @param weapon      The weapon instance to configure (must be a concrete Weapon subclass).
     * @param floorDepth  Current dungeon floor (1-based). Used for level clamping + tier weights.
     */
    public void rollWeapon(Weapon weapon, int floorDepth) {
        int        rolledLevel  = rollLevel(floorDepth);
        WeaponTier rolledTier   = rollTier(floorDepth);
        AbilityInstance[] abilities = rollAbilities(weapon, rolledTier, rolledLevel);
        weapon.configureRoll(rolledLevel, rolledTier, abilities);
    }

    /**
     * Configures a weapon as a fixed Common Level-1 run-start weapon (no randomness).
     */
    public void configureRunStart(Weapon weapon) {
        weapon.configureRoll(GameBalance.RUN_START_WEAPON_LEVEL,
                             GameBalance.RUN_START_WEAPON_TIER,
                             new AbilityInstance[0]);
    }

    // ── Private helpers ────────────────────────────────────────────────────

    private int rollLevel(int floorDepth) {
        int offset = rng.nextInt(3) - 1;  // -1, 0, or +1 uniformly
        int level  = floorDepth + offset;
        return Math.max(1, Math.min(WeaponConstants.MAX_WEAPON_LEVEL, level));
    }

    private WeaponTier rollTier(int floorDepth) {
        WeaponTier[] tiers = WeaponTier.values();
        float[] weights = new float[tiers.length];
        float totalWeight = 0f;

        for (int tierIndex = 0; tierIndex < tiers.length; tierIndex++) {
            float baseWeight  = getTierBaseWeight(tiers[tierIndex]);
            float drift       = getTierDrift(tiers[tierIndex]);
            float minFloor    = getTierMinFloor(tiers[tierIndex]);
            float cap         = getTierCap(tiers[tierIndex]);
            float computed    = baseWeight + drift * (floorDepth - 1);
            computed = Math.max(minFloor, computed);
            if (cap > 0) computed = Math.min(cap, computed);
            weights[tierIndex] = Math.max(0f, computed);
            totalWeight += weights[tierIndex];
        }

        float roll = rng.nextFloat() * totalWeight;
        float cumulative = 0f;
        for (int tierIndex = 0; tierIndex < tiers.length; tierIndex++) {
            cumulative += weights[tierIndex];
            if (roll < cumulative) return tiers[tierIndex];
        }
        return WeaponTier.COMMON;  // fallback
    }

    private float getTierBaseWeight(WeaponTier tier) {
        switch (tier) {
            case COMMON:    return WeaponConstants.TIER_WEIGHT_COMMON_BASE;
            case UNCOMMON:  return WeaponConstants.TIER_WEIGHT_UNCOMMON_BASE;
            case RARE:      return WeaponConstants.TIER_WEIGHT_RARE_BASE;
            case EPIC:      return WeaponConstants.TIER_WEIGHT_EPIC_BASE;
            case LEGENDARY: return WeaponConstants.TIER_WEIGHT_LEGENDARY_BASE;
            default:        return 0f;
        }
    }
    // ... getTierDrift, getTierMinFloor, getTierCap follow the same switch pattern ...

    private AbilityInstance[] rollAbilities(Weapon weapon, WeaponTier tier, int level) {
        if (tier.abilitySlots == 0) return new AbilityInstance[0];  // COMMON: no abilities

        // Collect eligible standard pool (legendaryOnly=false, eligibleFor this weapon)
        java.util.List<WeaponAbility> standardPool = new java.util.ArrayList<>();
        for (WeaponAbility ability : WeaponAbility.values()) {
            if (!ability.legendaryOnly && ability.eligibleFor(weapon.isMelee())) {
                standardPool.add(ability);
            }
        }

        int standardCount = tier.standardSlotCount();
        java.util.List<AbilityInstance> result = new java.util.ArrayList<>();

        // Sample without replacement from standard pool
        java.util.Collections.shuffle(standardPool, rng);
        for (int slotIndex = 0; slotIndex < Math.min(standardCount, standardPool.size()); slotIndex++) {
            result.add(buildInstance(standardPool.get(slotIndex), level));
        }

        // For Legendary: add one signature ability (legendaryOnly=true, eligible family)
        if (tier.isLegendary()) {
            java.util.List<WeaponAbility> legendaryPool = new java.util.ArrayList<>();
            for (WeaponAbility ability : WeaponAbility.values()) {
                if (ability.legendaryOnly && ability.eligibleFor(weapon.isMelee())) {
                    legendaryPool.add(ability);
                }
            }
            if (!legendaryPool.isEmpty()) {
                WeaponAbility signature = legendaryPool.get(rng.nextInt(legendaryPool.size()));
                result.add(buildInstance(signature, level));
            }
        }

        return result.toArray(new AbilityInstance[0]);
    }

    private AbilityInstance buildInstance(WeaponAbility ability, int weaponLevel) {
        float magnitude = 0f;
        int   countVal  = 0;

        switch (ability) {
            case CRITICAL_STRIKE:
                magnitude = GameMath.abilityMagnitude(GameBalance.CRIT_CHANCE_BASE,
                    GameBalance.CRIT_CHANCE_PER_LEVEL, weaponLevel, GameBalance.CRIT_CHANCE_CAP);
                break;
            case BURST_FIRE:
                countVal = GameMath.abilityCountScaled(2, 2, weaponLevel, 2, 5);
                magnitude = countVal;
                break;
            case LIFESTEAL:
                magnitude = GameMath.abilityMagnitude(GameBalance.LIFESTEAL_BASE,
                    GameBalance.LIFESTEAL_PER_LEVEL, weaponLevel, GameBalance.LIFESTEAL_CAP);
                break;
            case HEMORRHAGE_HARVEST:
                magnitude = GameMath.abilityMagnitude(GameBalance.HEMORRHAGE_HP_BASE,
                    GameBalance.HEMORRHAGE_HP_PER_LEVEL, weaponLevel, GameBalance.HEMORRHAGE_HP_CAP);
                break;
            // ... one case per ability using the matching GameBalance constants ...
            // Abilities with no numeric scaling (ARMOR_PIERCE base is PASSIVE magnitude):
            case ARMOR_PIERCE:
                magnitude = GameMath.abilityMagnitude(GameBalance.ARMOR_PIERCE_BASE,
                    GameBalance.ARMOR_PIERCE_PER_LEVEL, weaponLevel, GameBalance.ARMOR_PIERCE_CAP);
                break;
            case EXTENDED_MAG:
                countVal  = GameMath.abilityCountScaled(GameBalance.EXTENDED_MAG_BASE_COUNT,
                    GameBalance.EXTENDED_MAG_LEVELS_PER_STEP, weaponLevel,
                    GameBalance.EXTENDED_MAG_BASE_COUNT, GameBalance.EXTENDED_MAG_MAX_COUNT);
                magnitude = countVal;
                break;
            // Default: magnitude 0 (ability has no numeric parameter; it's boolean on/off)
            default:
                magnitude = 1f;  // "present" flag; each resolver case handles semantics
                break;
        }

        return new AbilityInstance(ability, magnitude, countVal);
    }
}

IMPLEMENTATION NOTES
====================
- The WeaponRoller's RNG is seeded ONCE from runSeed. Every subsequent call to
  rollWeapon() advances the RNG sequence. This is intentional: same runSeed →
  same sequence of weapon rolls → reproducible run (roguelike seed-share convention).
  Ensure runSeed is set before the first rollWeapon() call and is not changed mid-run.
- buildInstance() must have a case for EVERY ability in WeaponAbility enum. A
  default case handles any future abilities that have no magnitude. Consider using
  a compile-time check (IDE will warn on incomplete switch) to catch new abilities.
- The shuffle(standardPool, rng) consumes RNG state. Order of operations matters
  for reproducibility: always process pools in enum declaration order before
  shuffling so the shuffle is the only non-deterministic step.
- rollWeaponWithMinTier() for start-room offers: clamp the tier roll by re-rolling
  if the result is below minTier, or by zeroing out weights below minTier before
  normalizing. The re-roll approach is simpler but biases high; the weight-zeroing
  approach is cleaner for balance.
- Run `./gradlew build` after implementation. This is the last "logic" order;
  order-12 is purely display.
